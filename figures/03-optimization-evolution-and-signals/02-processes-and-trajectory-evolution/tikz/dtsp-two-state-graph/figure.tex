\begin{tikzpicture}[font=\figurefont\fontsize{8}{10}\selectfont,text=FigureInk,>=Stealth,
  st/.style={circle,draw=FigureBlue,fill=FigureBlueFill,line width=.75pt,minimum size=9mm,inner sep=0pt},
  st2/.style={circle,draw=FigureGreen,fill=FigureGreenFill,line width=.75pt,minimum size=9mm,inner sep=0pt},
