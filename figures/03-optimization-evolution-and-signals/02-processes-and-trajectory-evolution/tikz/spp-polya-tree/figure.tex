% Pólya 罐的逐步核构造。罐中初始有一个标 1 的球和一个标 0 的球；每次抽出一球，
% 记录其标号，放回并再加入一个同标号的球。边上数字是给定此前历史的下一步条件概率
% K_n(x_1..x_{n-1},{x_n})=(1+#1)/(2+n-1) 或其补；叶子写三步路径及其概率（乘积）。
% 含两个1的三条路径（黄色）概率都是1/12，含一个1的三条路径（绿色）也都是1/12。
\begin{tikzpicture}[line width=.6pt,font=\fontsize{8}{10}\selectfont,>=Stealth,
  dot/.style={circle,fill=FigureStroke,inner sep=0pt,minimum size=3pt},
  ep/.style={font=\fontsize{7.5}{9}\selectfont,text=FigureInk,fill=white,inner sep=.6pt},
  leaf/.style={anchor=west,inner sep=1.2pt,rounded corners=1.8pt}]
\def\dx{24mm}
\coordinate (r) at (0,0);
\coordinate (a1) at (\dx,13mm);   \coordinate (a0) at (\dx,-13mm);
\coordinate (b11) at (2*\dx,19.5mm); \coordinate (b10) at (2*\dx,6.5mm);
\coordinate (b01) at (2*\dx,-6.5mm); \coordinate (b00) at (2*\dx,-19.5mm);
\foreach \n/\y in {111/22.75,110/16.25,101/9.75,100/3.25,011/-3.25,010/-9.75,001/-16.25,000/-22.75}{
  \coordinate (c\n) at (3*\dx,\y mm);}
\draw[FigureStroke] (r) -- node[ep,pos=.55] {$1/2$} (a1);
\draw[FigureStroke] (r) -- node[ep,pos=.55] {$1/2$} (a0);
\draw[FigureStroke] (a1) -- node[ep,pos=.55] {$2/3$} (b11);
\draw[FigureStroke] (a1) -- node[ep,pos=.55] {$1/3$} (b10);
\draw[FigureStroke] (a0) -- node[ep,pos=.55] {$1/3$} (b01);
\draw[FigureStroke] (a0) -- node[ep,pos=.55] {$2/3$} (b00);
\foreach \p/\a/\b/\pa/\pb in {b11/111/110/{3/4}/{1/4}, b10/101/100/{1/2}/{1/2}, b01/011/010/{1/2}/{1/2}, b00/001/000/{1/4}/{3/4}}{
  \draw[FigureStroke] (\p) -- node[ep,pos=.68] {$\pa$} (c\a);
  \draw[FigureStroke] (\p) -- node[ep,pos=.68] {$\pb$} (c\b);}
\foreach \c in {r,a1,a0,b11,b10,b01,b00}{\node[dot] at (\c) {};}
\node[FigureMuted,font=\fontsize{7.5}{9}\selectfont,anchor=east] at (-1mm,0) {(1,1)};
\node[FigureMuted,font=\fontsize{7.5}{9}\selectfont,anchor=south] at ([yshift=1mm]a1) {(2,1)};
\node[FigureMuted,font=\fontsize{7.5}{9}\selectfont,anchor=north] at ([yshift=-1mm]a0) {(1,2)};
\node[leaf] at (c111) {111\quad $1/4$};
\node[leaf,fill=FigureYellowFill] at (c110) {110\quad $1/12$};
\node[leaf,fill=FigureYellowFill] at (c101) {101\quad $1/12$};
\node[leaf,fill=FigureGreenFill] at (c100) {100\quad $1/12$};
\node[leaf,fill=FigureYellowFill] at (c011) {011\quad $1/12$};
\node[leaf,fill=FigureGreenFill] at (c010) {010\quad $1/12$};
\node[leaf,fill=FigureGreenFill] at (c001) {001\quad $1/12$};
\node[leaf] at (c000) {000\quad $1/4$};
\node[FigureInk] at (\dx,-28mm) {$X_1$};
\node[FigureInk] at (2*\dx,-28mm) {$X_2$};
\node[FigureInk] at (3*\dx,-28mm) {$X_3$};
\end{tikzpicture}
