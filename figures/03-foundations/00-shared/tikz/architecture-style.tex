   minimum width=#4mm,minimum height=#4mm,inner sep=0pt] (#1-outer) at (#2,#3) {};
 \node[circle,draw=#5,fill=#5!32!white,line width=.9pt,
   minimum size=\VOperatorCore mm,inner sep=.4pt,
