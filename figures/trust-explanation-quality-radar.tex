% 构造的评价方案覆盖评分，非模型/解释算法性能。每轴预设4项检查。
% 甲纳入(4,2,1)项、乙纳入(2,4,3)项；评分仅为纳入项数/4，不聚合认证。
\begin{tikzpicture}[foundation picture,
 font=\figurefont\fontsize{8.5}{11}\selectfont,
 foundation heading/.append style={font=\figurefont\mdseries\fontsize{8.5}{11}\selectfont},
 foundation note/.append style={font=\figurefont\fontsize{8.5}{11}\selectfont}]
 \path[use as bounding box] (0,-2) rectangle(110,73);
 \begin{scope}[shift={(55,37)}]
  \foreach \q in {.25,.5,.75,1}{
   \draw[FigureGrid,line width=.5pt]
    ({30*\q*cos(90)},{30*\q*sin(90)})--
    ({30*\q*cos(210)},{30*\q*sin(210)})--
    ({30*\q*cos(330)},{30*\q*sin(330)})--cycle;
  }
  \foreach \a in {90,210,330}{
   \draw[FigureStroke,line width=.6pt] (0,0)--({30*cos(\a)},{30*sin(\a)});
  }
  \node[anchor=south] at (0,31) {忠实性};
  \node[anchor=north east] at (-27,-16) {覆盖范围};
  \node[anchor=north west] at (27,-16) {稳定性};
  \node[anchor=west,fill=white,inner sep=1pt] at (4,30) {$1$};
  \node[anchor=west,fill=white,inner sep=1pt] at (4,15) {$0.5$};
  \path[fill=FigureBlueFill] (0,30)--({15*cos(210)},{15*sin(210)})--({7.5*cos(330)},{7.5*sin(330)})--cycle;
  \draw[FigureBlue,line width=1pt] (0,30)--({15*cos(210)},{15*sin(210)})--({7.5*cos(330)},{7.5*sin(330)})--cycle;
  \foreach \a/\r in {90/30,210/15,330/7.5}{
   \fill[FigureBlue] ({\r*cos(\a)},{\r*sin(\a)}) circle(1);
  }
  \draw[FigureRed,line width=1pt,dashed] (0,15)--({30*cos(210)},{30*sin(210)})--({22.5*cos(330)},{22.5*sin(330)})--cycle;
  \foreach \a/\r in {90/15,210/30,330/22.5}{
   \draw[FigureRed,fill=white,line width=.9pt]
    ({\r*cos(\a)-.9},{\r*sin(\a)-.9}) rectangle({\r*cos(\a)+.9},{\r*sin(\a)+.9});
  }
 \end{scope}
 \draw[FigureBlue,line width=1pt] (30,5)--(39,5);
 \node[anchor=west] at (41,5) {甲};
 \draw[FigureRed,line width=1pt,dashed] (66,5)--(75,5);
 \node[anchor=west] at (77,5) {乙};
\end{tikzpicture}
