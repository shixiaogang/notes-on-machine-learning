  % 输入：窗口内的n-1=4个词编号
  \foreach \i/\w in {0/{t-4},1/{t-3},2/{t-2},3/{t-1}}{
    \node[index] (w\i) at (\i*17,0) {$x_{\w}$};

  \node[shape note] at (62,72) {对全部$V$个得分归一化};
