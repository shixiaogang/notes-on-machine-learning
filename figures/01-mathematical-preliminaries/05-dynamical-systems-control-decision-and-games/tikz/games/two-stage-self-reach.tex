% A single participant acts at I_0 and then, after E, at I_1.
% Terminal rewards: Q -> 0, EL -> 1, ER -> 0. No chance or opponent nodes.
% The 112 mm tree reads top to bottom; short edge labels identify action
% probabilities, while circle/square shapes distinguish decision/terminal nodes.
\begin{tikzpicture}[figure picture,x=1mm,y=1mm,
  decision/.style={circle,draw=FigureBlue,fill=FigureBlue!8,
    line width=.8pt,minimum size=8mm,inner sep=1pt},
  terminal/.style={rectangle,draw=FigureStroke,fill=FigureYellow!15,
    line width=.8pt,minimum size=7mm,inner sep=1pt},
  branch/.style={draw=FigureStroke,line width=.9pt,
    -{Stealth[length=1.8mm,width=1.2mm]}}]
  \path[use as bounding box] (0,0) rectangle (112,52);
  \node[decision] (root) at (48,45) {$I_0$};
  \node[terminal] (quit) at (17,27) {$0$};
  \node[decision] (next) at (79,27) {$I_1$};
  \node[terminal] (left) at (64,6) {$1$};
  \node[terminal] (right) at (97,6) {$0$};
  \draw[branch] (root)--(quit)
    node[pos=.5,above left] {退出$Q$，$1-p$};
  \draw[branch] (root)--(next)
    node[pos=.5,above right] {进入$E$，$p$};
  \draw[branch] (next)--(left) node[pos=.5,left] {$L$，$q$};
  \draw[branch] (next)--(right) node[pos=.5,right] {$R$，$1-q$};
\end{tikzpicture}
