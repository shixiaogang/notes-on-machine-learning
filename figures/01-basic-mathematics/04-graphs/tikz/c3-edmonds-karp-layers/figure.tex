% Edmonds--Karp 三轮（Python 验算，邻居按字母序）：
% 轮1 层 a=1,b=1,t=2，路径 s-a-t 瓶颈 2；轮2 层同，路径 s-b-t 瓶颈 2；轮3 层 a=1,b=2,t=3，路径 s-a-b-t 瓶颈 1。
\begin{tikzpicture}[>=Stealth,
  v/.style={circle,draw=FigureStroke,fill=FigureBlueFill,line width=.8pt,minimum size=6mm,inner sep=0pt},
  fw/.style={->,draw=FigureBlue,line width=.8pt},
  bk/.style={->,draw=FigureGreen,dashed,line width=.8pt},
  aug/.style={->,draw=FigureYellow,line width=1.8pt},
  w/.style={fill=white,inner sep=.5pt,font=\figurefont\fontsize{7}{8}\selectfont},
  lab/.style={font=\figurefont\fontsize{7.5}{9}\selectfont},
  lev/.style={font=\figurefont\fontsize{7}{8}\selectfont,text=FigureMuted}]
% 第 1 轮
\begin{scope}
  \coordinate (S) at (0,0); \coordinate (A) at (1.3,.9); \coordinate (B) at (1.3,-.9); \coordinate (T) at (2.6,0);
  \node[v] (s) at (S) {$s$}; \node[v] (a) at (A) {$a$}; \node[v] (b) at (B) {$b$}; \node[v] (t) at (T) {$t$};
  \draw[aug] (s)--node[w]{$3$}(a); \draw[fw] (s)--node[w]{$2$}(b); \draw[fw] (a)--node[w]{$1$}(b);
  \draw[aug] (a)--node[w]{$2$}(t); \draw[fw] (b)--node[w]{$3$}(t);
  \foreach \x/\h in {0/0,1.3/1,2.6/2} \node[lev] at (\x,-1.55) {$h=\h$};
  \node[lab,align=center] at (1.3,-2.15) {(1) 路径$s\to a\to t$，瓶颈$2$};
\end{scope}
% 第 2 轮
\begin{scope}[xshift=3.7cm]
  \coordinate (S) at (0,0); \coordinate (A) at (1.3,.9); \coordinate (B) at (1.3,-.9); \coordinate (T) at (2.6,0);
  \node[v] (s) at (S) {$s$}; \node[v] (a) at (A) {$a$}; \node[v] (b) at (B) {$b$}; \node[v] (t) at (T) {$t$};
  \draw[fw] (s) to[bend left=18] node[w]{$1$}(a); \draw[bk] (a) to[bend left=18] node[w]{$2$}(s);
  \draw[aug] (s)--node[w]{$2$}(b); \draw[fw] (a)--node[w]{$1$}(b);
  \draw[bk] (t)--node[w]{$2$}(a); \draw[aug] (b)--node[w]{$3$}(t);
  \foreach \x/\h in {0/0,1.3/1,2.6/2} \node[lev] at (\x,-1.55) {$h=\h$};
  \node[lab,align=center] at (1.3,-2.15) {(2) 路径$s\to b\to t$，瓶颈$2$};
\end{scope}
% 第 3 轮
\begin{scope}[xshift=7.4cm]
  \coordinate (S) at (0,0); \coordinate (A) at (1.2,.75); \coordinate (B) at (2.4,-.75); \coordinate (T) at (3.6,0);
  \node[v] (s) at (S) {$s$}; \node[v] (a) at (A) {$a$}; \node[v] (b) at (B) {$b$}; \node[v] (t) at (T) {$t$};
  \draw[aug] (s) to[bend left=18] node[w]{$1$}(a); \draw[bk] (a) to[bend left=18] node[w]{$2$}(s);
  \draw[bk] (b) to[bend left=12] node[w]{$2$}(s);
  \draw[aug] (a)--node[w]{$1$}(b);
  \draw[bk] (t) to[bend right=18] node[w]{$2$}(a);
  \draw[aug] (b) to[bend left=18] node[w]{$1$}(t); \draw[bk] (t) to[bend left=18] node[w]{$2$}(b);
  \foreach \x/\h in {0/0,1.2/1,2.4/2,3.6/3} \node[lev] at (\x,-1.55) {$h=\h$};
  \node[lab,align=center] at (1.8,-2.15) {(3) 路径$s\to a\to b\to t$，瓶颈$1$};
\end{scope}
\end{tikzpicture}
