% 第6章 6.5.2：八点单位权环的复指数特征向量 u_k[r]=e^{2 pi i k r/8}/sqrt8，r=0,...,7。
% 实部 cos(2 pi k r/8)/sqrt8（蓝色实心），虚部 sin(2 pi k r/8)/sqrt8（绿色空心）；
% 特征值 lambda_k=2-2cos(2 pi k/8)：0, 0.586, 2, 3.414, 4, 3.414, 2, 0.586。
% 曲线由 pgfplots 按解析式在整数点求值，1/sqrt8=0.35355。
\begin{tikzpicture}[font=\figurefont\footnotesize]
\pgfplotsset{every axis/.style={width=34mm,height=24mm,scale only axis=false,
  xmin=-0.6,xmax=7.6,ymin=-0.42,ymax=0.42,axis lines=left,
  xtick={0,2,4,6},ytick=\empty,
  x axis line style={draw=FigureStroke,line width=.5pt,-},
  y axis line style={draw=none},
  tick style={draw=FigureStroke},tick label style={font=\figurefont\scriptsize,text=FigureInk},
  extra y ticks={0},extra y tick style={grid=major,major grid style={draw=FigureGrid,line width=.5pt},tick label style={opacity=0},tick style={draw=none}},
  title style={font=\figurefont\scriptsize,yshift=-1.2mm},
  samples at={0,1,...,7}}}
\foreach \k/\lam [count=\i from 0] in {0/0,1/0.586,2/2,3/3.414,4/4,5/3.414,6/2,7/0.586}{
  \pgfmathsetmacro{\col}{mod(\i,4)}
  \pgfmathsetmacro{\row}{int(\i/4)}
  \begin{axis}[at={(\col*29mm,-\row*21mm)},anchor=south west,
     title={$k=\k$，$\lambda_{\k}=\lam$}]
   \addplot[ycomb,draw=FigureBlue,line width=.8pt,mark=*,mark size=1.1pt,mark options={fill=FigureBlue}] {0.35355*cos(360*\k*x/8)};
   \addplot[ycomb,draw=FigureGreen,line width=.6pt,mark=o,mark size=1.3pt,mark options={draw=FigureGreen,fill=white},xshift=0.9pt] {0.35355*sin(360*\k*x/8)};
  \end{axis}}
\node[anchor=west,font=\figurefont\scriptsize] at (0,-27mm) {横轴：顶点编号$r$；};
\draw[FigureBlue,line width=.8pt] (24mm,-27mm)--(27mm,-27mm); \fill[FigureBlue] (25.5mm,-27mm) circle (1.1pt);
\node[anchor=west,font=\figurefont\scriptsize] at (27.5mm,-27mm) {实部};
\draw[FigureGreen,line width=.6pt] (37mm,-27mm)--(40mm,-27mm); \draw[FigureGreen,fill=white] (38.5mm,-27mm) circle (1.3pt);
\node[anchor=west,font=\figurefont\scriptsize] at (40.5mm,-27mm) {虚部};
\end{tikzpicture}
