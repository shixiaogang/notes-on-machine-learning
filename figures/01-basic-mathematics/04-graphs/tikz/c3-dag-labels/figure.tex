% DAG 最短路：s->a 2, s->b 5, a->b -1, a->t 5, b->t 2，拓扑序 (s,a,b,t)。
% 标签变化：a: inf->2；b: inf->5->1；t: inf->7->3（Python 验算）。
\begin{tikzpicture}[>=Stealth,
  v/.style={circle,draw=FigureStroke,fill=FigureBlueFill,line width=.8pt,minimum size=6.5mm,inner sep=0pt},
  arc/.style={->,draw=FigureStroke,line width=.8pt},
  best/.style={->,draw=FigureBlue,line width=1.6pt},
  w/.style={fill=white,inner sep=.8pt,font=\figurefont\fontsize{7.5}{9}\selectfont},
  lab/.style={font=\figurefont\fontsize{7.5}{9}\selectfont},
  note/.style={font=\FigureNoteFont\fontsize{7.5}{9}\selectfont}]
  \node[v] (s) at (0,0) {$s$};
  \node[v] (a) at (2,1.1) {$a$};
  \node[v] (b) at (2,-1.1) {$b$};
  \node[v] (t) at (4,0) {$t$};
  \draw[best] (s)--node[w]{$2$}(a);
  \draw[arc] (s)--node[w]{$5$}(b);
  \draw[best] (a)--node[w]{$-1$}(b);
  \draw[arc] (a)--node[w]{$5$}(t);
  \draw[best] (b)--node[w]{$2$}(t);
  \node[lab,left=1mm of s] {$0$};
  \node[lab,above=1mm of a] {$+\infty\to2$};
  \node[lab,below=1mm of b] {$+\infty\to5\to1$};
  \node[lab,right=1mm of t] {$+\infty\to7\to3$};
  \node[note,text=FigureMuted] at (2,-2.15) {拓扑序：$s,a,b,t$};
\end{tikzpicture}
