% 第17章17.4.2匹配的单位容量网络：源 s 到左侧 a,b，允许边 (a,1),(a,2),(b,1) 改为左到右的弧，右侧 1,2 到汇 t，容量均为1。
% 匹配 {(a,1)} 对应流值1的整数流（蓝）；残量路径 s->b->1->a->2->t（绿）中 1->a 是撤回匹配边 (a,1) 的反向残量弧。
\begin{tikzpicture}[>=Stealth,line width=.8pt,font=\fontsize{8}{10}\selectfont,
  v/.style={circle,draw=FigureStroke,fill=FigureBlueFill,line width=.8pt,minimum size=6.5mm,inner sep=0pt,text=FigureInk},
  e/.style={->,draw=FigureMuted,line width=.8pt,shorten >=.5pt},
  f/.style={->,draw=FigureBlue,line width=1.6pt,shorten >=.5pt},
  p/.style={->,draw=FigureGreen,line width=1.4pt,dashed,shorten >=.5pt},
  note/.style={text=FigureMuted,font=\FigureNoteFont\fontsize{7.5}{9}\selectfont}]
  \node[v] (s) at (0,0) {$s$};
  \node[v] (a) at (1.9,-.75) {$a$};
  \node[v] (b) at (1.9,.75) {$b$};
  \node[v] (x) at (4.3,.75) {$1$};
  \node[v] (y) at (4.3,-.75) {$2$};
  \node[v] (t) at (6.2,0) {$t$};
  \draw[f] (s)--(a); \draw[e] (s)--(b);
  \draw[f] (a)--(x); \draw[e] (a)--(y); \draw[e] (b)--(x);
  \draw[f] (x)--(t); \draw[e] (y)--(t);
  % 残量路径
  \draw[p] (s) to[bend left=24] (b);
  \draw[p] (b) to[bend left=18] (x);
  \draw[p] (x) to[bend left=12] (a);
  \draw[p] (a) to[bend right=18] (y);
  \draw[p] (y) to[bend right=24] (t);
  \node[note] at (3.1,1.55) {反向残量弧$1\to a$表示撤回匹配边$(a,1)$};
  \node[note] at (3.1,-1.55) {所有弧容量为$1$；蓝色粗线：当前流；绿色虚线：残量增广路径};
\end{tikzpicture}
