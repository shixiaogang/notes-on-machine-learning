% 第17章17.5 KM算法算例：成本 C=[[1,2],[1,4]]（行：甲、乙；列：任务1、任务2）。
% (a) u=(1,1), v=(0,0)：剩余成本 [[0,1],[0,3]]；甲配任务1；从自由的乙建交替树：乙->任务1->甲，受阻。
%     S={甲,乙}，T={任务1}；任务2 的最小差额 min(1,3)=1，来自甲，故 delta=1。
% (b) u=(2,2), v=(-1,0)：剩余成本 [[0,0],[0,2]]；甲->任务2 变紧；增广路径 乙-任务1-甲-任务2。
% (c) 匹配 {甲-任务2, 乙-任务1}，成本 2+1=3 = 2+2-1+0。逐行贪心为 1+4=5。
% 数值由 scratchpad 中的 verify.py 复算。
\begin{tikzpicture}[line width=.8pt,font=\fontsize{8}{10}\selectfont,
  rw/.style={rectangle,rounded corners=1.8pt,draw=FigureStroke,fill=FigureBlueFill,line width=.8pt,minimum width=6.5mm,minimum height=6mm,inner sep=1pt,text=FigureInk},
  cl/.style={rw,fill=FigureGreenFill,minimum width=10mm},
  tight/.style={draw=FigureStroke,line width=.9pt},
  slack/.style={draw=FigureMuted,line width=.7pt,dashed},
  m/.style={draw=FigureBlue,line width=2pt},
  tree/.style={draw=FigureYellow,line width=3.6pt,opacity=.55},
  aug/.style={draw=FigureGreen,line width=1.6pt},
  lab/.style={text=FigureInk,inner sep=1.2pt,fill=white},
  pot/.style={text=FigureMuted,font=\FigureNoteFont\fontsize{7.5}{9}\selectfont}]
  \foreach \dx/\k in {0/A,3.55/B,7.1/C}{
    \node[rw] (r1\k) at (\dx,.65) {甲};
    \node[rw] (r2\k) at (\dx,-.65) {乙};
    \node[cl] (c1\k) at (\dx+2.1,.65) {任务$1$};
    \node[cl] (c2\k) at (\dx+2.1,-.65) {任务$2$};
  }
  % (a)
  \draw[tree] (r2A)--(c1A); \draw[tree] (r1A)--(c1A);
  \draw[m] (r1A)--node[lab,pos=.25]{$0$} (c1A);
  \draw[tight] (r2A)--node[lab,pos=.25]{$0$} (c1A);
  \draw[slack] (r1A)--node[lab,pos=.25]{$1$} (c2A);
  \draw[slack] (r2A)--node[lab,pos=.25]{$3$} (c2A);
  \node[pot,above=1pt] at (r1A.north) {$u=1$};
  \node[pot,below=1pt] at (r2A.south) {$u=1$};
  \node[pot,above=1pt] at (c1A.north) {$v=0$};
  \node[pot,below=1pt] at (c2A.south) {$v=0$};
  \node at (1.05,-1.9) {(a) 交替树受阻，$\delta=1$};
  % (b)
  \draw[aug] (r2B)--node[lab,pos=.25]{$0$} (c1B);
  \draw[m] (r1B)--(c1B);
  \draw[aug,dashed] (r1B)--(c1B);
  \draw[aug] (r1B)--node[lab,pos=.25]{$0$} (c2B);
  \draw[slack] (r2B)--node[lab,pos=.25]{$2$} (c2B);
  \node[pot,above=1pt] at (r1B.north) {$u=2$};
  \node[pot,below=1pt] at (r2B.south) {$u=2$};
  \node[pot,above=1pt] at (c1B.north) {$v=-1$};
  \node[pot,below=1pt] at (c2B.south) {$v=0$};
  \node at (4.6,-1.9) {(b) 调标后出现增广路径};
  % (c)
  \draw[tight] (r1C)--(c1C);
  \draw[m] (r2C)--node[lab,pos=.25]{$1$} (c1C);
  \draw[m] (r1C)--node[lab,pos=.25]{$2$} (c2C);
  \draw[slack] (r2C)--(c2C);
  \node[pot,above=1pt] at (r1C.north) {$u=2$};
  \node[pot,below=1pt] at (r2C.south) {$u=2$};
  \node[pot,above=1pt] at (c1C.north) {$v=-1$};
  \node[pot,below=1pt] at (c2C.south) {$v=0$};
  \node at (8.15,-1.9) {(c) 最优指派，成本$3$};
\end{tikzpicture}
