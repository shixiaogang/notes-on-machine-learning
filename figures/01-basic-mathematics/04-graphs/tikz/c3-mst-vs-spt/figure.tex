% 最小生成树与最短路树：xy=2, yz=2, xz=3。MST {xy,yz} 成本 4；以 x 为源的最短路树 {xy,xz} 成本 5。
\begin{tikzpicture}[>=Stealth,
  v/.style={circle,draw=FigureStroke,fill=FigureBlueFill,line width=.8pt,minimum size=6.5mm,inner sep=0pt},
  e/.style={draw=FigureGrid,line width=.8pt},
  h/.style={draw=FigureBlue,line width=1.6pt},
  w/.style={fill=white,inner sep=.8pt,font=\figurefont\fontsize{7.5}{9}\selectfont},
  lab/.style={font=\figurefont\fontsize{7.5}{9}\selectfont}]
\newcommand{\tri}{\node[v] (x) at (0,0) {$x$}; \node[v] (y) at (1,1.3) {$y$}; \node[v] (z) at (2,0) {$z$};}
\begin{scope}
  \tri
  \draw[draw=FigureStroke,line width=.8pt] (x)--node[w]{$2$}(y) (y)--node[w]{$2$}(z) (x)--node[w]{$3$}(z);
  \node[lab,align=center] at (1,-.85) {(a) 原图};
\end{scope}
\begin{scope}[xshift=3.3cm]
  \tri
  \draw[e] (x)--node[w]{$3$}(z);
  \draw[h] (x)--node[w]{$2$}(y) (y)--node[w]{$2$}(z);
  \node[lab,align=center] at (1,-.85) {(b) 最小生成树，成本$4$\\树上$x$到$z$长$4$};
\end{scope}
\begin{scope}[xshift=6.6cm]
  \tri
  \draw[e] (y)--node[w]{$2$}(z);
  \draw[h] (x)--node[w]{$2$}(y) (x)--node[w]{$3$}(z);
  \node[lab,align=center] at (1,-.85) {(c) 以$x$为源的最短路树\\成本$5$，$x$到$z$长$3$};
\end{scope}
\end{tikzpicture}
