\begin{tikzpicture}[x=9mm,y=9mm,line width=.6pt,font=\figurefont\fontsize{8}{10}\selectfont]
  % 格子
  \foreach \c in {1,...,8}{
    \foreach \r/\y in {1/1,2/0}{
      \draw[FigureGrid,line width=.5pt,fill=white] (\c-1,\y) rectangle (\c,\y+1);
    }
  }
  % 覆盖次数
  \foreach \c/\top/\bot in {1/2/1,2/2/2,3/2/2,4/2/2,5/2/2,6/2/2,7/1/1,8/1/1}{

    \node[FigureMuted,font=\FigureNoteFont\fontsize{7.5}{9}\selectfont] at (\c-.5,.5) {\bot};
  }
  % 传感器范围
  \draw[FigureGreen,line width=1.1pt,rounded corners=1.5pt] (0.08,0.08) rectangle (3.92,1.92);

  \node[FigureBlue,anchor=south] at (3.5,2.25) {$a$：10格};

  \node[FigurePurple,anchor=east] at (-0.15,1.5) {$d$：1格};
  \foreach \c in {1,...,8}{\node[FigureInk] at (\c-.5,-0.5) {\c};}
  \node[FigureInk,anchor=east] at (-0.15,-0.5) {列号};
