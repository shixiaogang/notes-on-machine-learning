% 第2章 2.1.1：u_n(x)=sin(nx) 在一致范数下有界，而 Du_n=n cos(nx) 的一致范数为 n。
% 解析式直接绘制，n=1（蓝实线）与 n=5（绿虚线），区间 [0,2π]。
\begin{tikzpicture}[font=\figurefont\footnotesize]
\pgfplotsset{every axis/.style={width=56mm,height=36mm,axis lines=left,
  x axis line style={draw=FigureStroke,line width=.6pt},y axis line style={draw=FigureStroke,line width=.6pt},
  tick style={draw=FigureStroke},tick label style={font=\figurefont\scriptsize,text=FigureInk},
  label style={font=\figurefont\scriptsize},xlabel={$x$},
  xlabel style={at={(axis description cs:1,0)},anchor=north west,yshift=1mm},
  xmin=0,xmax=6.4,xtick={0,3.1416,6.2832},xticklabels={$0$,$\pi$,$2\pi$},
  ymajorgrids,grid style={draw=FigureGrid,line width=.4pt},
  title style={font=\figurefont\footnotesize},samples=300,domain=0:6.2832,
  legend style={draw=none,fill=none,font=\figurefont\scriptsize,cells={anchor=west}}}}
\begin{axis}[name=A,ymin=-1.3,ymax=1.3,ytick={-1,0,1},title={(a) $u_n(x)=\sin(nx)$},
  legend style={at={(0.5,-0.3)},anchor=north},legend columns=2]
\addplot[draw=FigureBlue,line width=1.1pt] {sin(deg(x))};
\addplot[draw=FigureGreen,line width=1.1pt,dashed] {sin(deg(5*x))};
\legend{$n=1$,$n=5$}
\end{axis}
\begin{axis}[at={(A.east)},anchor=west,xshift=10mm,ymin=-5.8,ymax=5.8,ytick={-5,0,5},title={(b) $Du_n(x)=n\cos(nx)$},
  legend style={at={(0.5,-0.3)},anchor=north},legend columns=2]
\addplot[draw=FigureBlue,line width=1.1pt] {cos(deg(x))};
\addplot[draw=FigureGreen,line width=1.1pt,dashed] {5*cos(deg(5*x))};
\legend{$n=1$,$n=5$}
\end{axis}
\end{tikzpicture}
