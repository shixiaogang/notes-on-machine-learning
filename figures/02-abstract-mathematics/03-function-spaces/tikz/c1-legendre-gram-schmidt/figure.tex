% 第1章 1.6.2：L^2([-1,1]) 中对 1,x,x^2,x^3 做 Gram--Schmidt。
% (a) 单项式；(b) 归一化 Legendre 多项式 e_0=1/sqrt2，e_1=sqrt(3/2)x，e_2=sqrt(5/8)(3x^2-1)，e_3=sqrt(7/8)(5x^3-3x)。
\begin{tikzpicture}[font=\figurefont\footnotesize]
\pgfplotsset{every axis/.style={width=50mm,height=40mm,axis lines=middle,
  x axis line style={draw=FigureStroke,line width=.6pt},y axis line style={draw=FigureStroke,line width=.6pt},
  tick style={draw=FigureStroke},tick label style={font=\figurefont\scriptsize,text=FigureInk},
  xmin=-1.05,xmax=1.12,xtick={-1,1},xlabel={$x$},xlabel style={anchor=north west,font=\figurefont\scriptsize},
  title style={font=\figurefont\footnotesize},samples=120,
  legend style={draw=none,fill=none,font=\figurefont\scriptsize,cells={anchor=west}}}}
\begin{axis}[name=A,ymin=-1.15,ymax=1.25,ytick={-1},title={(a) $1,x,x^2,x^3$},legend columns=2,legend style={at={(0.5,-0.08)},anchor=north}]
\addplot[draw=FigureBlue,line width=1.1pt,domain=-1:1] {1};
\addplot[draw=FigureGreen,line width=1.1pt,domain=-1:1] {x};
\addplot[draw=FigureYellow,line width=1.1pt,domain=-1:1] {x^2};
\addplot[draw=FigurePurple,line width=1.1pt,domain=-1:1] {x^3};
\legend{$1$,$x$,$x^2$,$x^3$}
\end{axis}
\begin{axis}[at={(A.east)},anchor=west,xshift=12mm,ymin=-2.0,ymax=2.0,ytick={-1,1},title={(b) 正交归一化后的 $e_0,\ldots,e_3$},legend columns=2,legend style={at={(0.5,-0.08)},anchor=north}]
\addplot[draw=FigureBlue,line width=1.1pt,domain=-1:1] {1/sqrt(2)};
\addplot[draw=FigureGreen,line width=1.1pt,domain=-1:1] {sqrt(1.5)*x};
\addplot[draw=FigureYellow,line width=1.1pt,domain=-1:1] {sqrt(5/8)*(3*x^2-1)};
\addplot[draw=FigurePurple,line width=1.1pt,domain=-1:1] {sqrt(7/8)*(5*x^3-3*x)};
\legend{$e_0$,$e_1$,$e_2$,$e_3$}
\end{axis}
\end{tikzpicture}
