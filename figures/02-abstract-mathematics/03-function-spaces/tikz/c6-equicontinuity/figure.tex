% 第8章 8.3.1：共同连续性尺度。窗口 [x_0-delta,x_0+delta]，x_0=0.4，delta=0.05（蓝底）。
% (a) f_a(x)=0.5+0.4 sin(2 pi x+a)，a=0,1,2,3：Lipschitz 常数均为 0.8 pi≈2.51，窗口内振幅至多 0.25；
% (b) g_n(x)=0.5+0.5 sin(2 pi n x)，n=1,4,16：窗口内 g_16 已扫过 [0,1]，族不等度连续。解析式绘制。
\begin{tikzpicture}[font=\figurefont\footnotesize]
\pgfplotsset{every axis/.style={axis lines=left,
  x axis line style={draw=FigureStroke,line width=.6pt},y axis line style={draw=FigureStroke,line width=.6pt},
  tick style={draw=FigureStroke},tick label style={font=\figurefont\scriptsize,text=FigureInk},
  title style={font=\figurefont\footnotesize},label style={font=\figurefont\scriptsize},
  legend style={draw=none,fill=none,font=\figurefont\scriptsize,cells={anchor=west}}}}
\pgfplotsset{every axis/.append style={width=52mm,height=36mm,xmin=0,xmax=1,ymin=0,ymax=1.05,xtick={0,0.4,1},ytick={0,0.5,1}}}
\begin{axis}[name=A,title={(a) 等度连续}]
\fill[FigureBlueFill] (axis cs:0.35,0) rectangle (axis cs:0.45,1.05);
\foreach \a/\s in {0/solid,1/dashed,2/dash dot,3/densely dotted}{
  \edef\tmp{\noexpand\addplot[draw=FigureBlue,line width=.9pt,\s,domain=0:1,samples=150] {0.5+0.4*sin(deg(2*pi*x+\a))};}\tmp}
\end{axis}
\begin{axis}[at={(A.east)},anchor=west,xshift=12mm,title={(b) 不等度连续}]
\fill[FigureBlueFill] (axis cs:0.35,0) rectangle (axis cs:0.45,1.05);
\addplot[draw=FigureMuted,line width=.8pt,domain=0:1,samples=150] {0.5+0.5*sin(deg(2*pi*x))};
\addplot[draw=FigureGreen,line width=.8pt,dashed,domain=0:1,samples=200] {0.5+0.5*sin(deg(8*pi*x))};
\addplot[draw=FigureBlue,line width=.7pt,domain=0:1,samples=600] {0.5+0.5*sin(deg(32*pi*x))};
\end{axis}
\end{tikzpicture}
