% 第3章 3.3.1：核 K(x,z)=1+xz 及其积分算子在 L^2([-1,1],dz/2) 上的作用。
% (a) K 在 [-1,1]^2 上的热图（取值 0 到 2），由解析式 1+x*z 直接取样（41x41 网格，平面着色）。
% (b) f=1+x 的像 Tf=1+x/3（int f dmu=1，int z f dmu=1/3）；
%     q=x^2-1/3 满足 int q dmu=0、int z q dmu=0，因此 Tq=0。
\begin{tikzpicture}[font=\figurefont\footnotesize]
\pgfplotsset{colormap={bluefill}{color(0)=(white) color(1)=(FigureBlueFill) color(2)=(FigureBlue)}}
\begin{axis}[name=A,width=43mm,height=43mm,
  view={0}{90},xmin=-1,xmax=1,ymin=-1,ymax=1,
  xtick={-1,0,1},ytick={-1,0,1},
  xlabel={$z$},ylabel={$x$},
  xlabel style={font=\figurefont\scriptsize},ylabel style={font=\figurefont\scriptsize},
  tick label style={font=\figurefont\scriptsize,text=FigureInk},
  axis line style={draw=FigureStroke,line width=.6pt},
  colorbar,colorbar style={width=2mm,ytick={0,1,2},tick label style={font=\figurefont\scriptsize}},
  point meta min=0,point meta max=2,
  title={(a) $K(x,z)=1+xz$},title style={font=\figurefont\footnotesize}]
\addplot3[surf,shader=flat corner,draw=none,samples=41,domain=-1:1,y domain=-1:1] {1+x*y};
\end{axis}
\begin{axis}[at={(A.east)},anchor=west,xshift=19mm,width=46mm,height=43mm,
  xmin=-1,xmax=1,ymin=-0.5,ymax=2.1,domain=-1:1,samples=60,
  axis x line=bottom,axis y line=left,
  xtick={-1,0,1},ytick={0,1,2},
  extra y ticks={0},extra y tick style={grid=major,major grid style={draw=FigureStroke,line width=.5pt}},
  xlabel={$x$},xlabel style={font=\figurefont\scriptsize},
  tick label style={font=\figurefont\scriptsize,text=FigureInk},
  ymajorgrids,grid style={draw=FigureGrid,line width=.4pt},
  legend style={at={(1.03,0.5)},anchor=west,font=\figurefont\scriptsize,cells={anchor=west}},
  title={(b) 积分算子的作用},title style={font=\figurefont\footnotesize}]
\addplot[FigureBlue,line width=1.1pt] {1+x};
\addplot[FigureBlue,line width=1.1pt,dashed] {1+x/3};
\addplot[FigureGreen,line width=1.1pt] {x^2-1/3};
\addplot[FigureGreen,line width=1.4pt,dashed] {0};
\legend{$f=1+x$,$Tf=1+x/3$,$q=x^2-1/3$,$Tq=0$}
\end{axis}
\end{tikzpicture}
