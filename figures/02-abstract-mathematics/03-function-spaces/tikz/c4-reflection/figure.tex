% 第4章 4.4.2：半区间上的两种延拓（解析式绘制）。u(x)=1+x-x^2（x∈[0,1]，u(0)=1）。
% (a) 偶反射 Eu(x)=u(|x|)：连续，原点处为尖点，弱导数 sgn(x)u'(|x|) 局部可积；(b) 零延拓：原点处跳跃 u(0)=1，导数含 delta_0 项。
\begin{tikzpicture}[font=\figurefont\footnotesize]
\pgfplotsset{every axis/.style={width=50mm,height=34mm,axis lines=middle,
  x axis line style={draw=FigureStroke,line width=.6pt,-{Latex[length=1.6mm]}},
  y axis line style={draw=FigureStroke,line width=.6pt,-{Latex[length=1.6mm]}},
  tick style={draw=FigureStroke},tick label style={font=\figurefont\scriptsize,text=FigureInk},
  xlabel={$x$},xlabel style={anchor=west,font=\figurefont\scriptsize},
  title style={font=\figurefont\footnotesize,yshift=1mm},xmin=-1.2,xmax=1.3,ymin=-0.15,ymax=1.5,xtick={-1,0,1},ytick={0,1},axis y line=left}}
\begin{axis}[name=A,title={(a) 偶反射}]
\addplot[draw=FigureBlue,line width=1.1pt,domain=0:1,samples=60] {1+x-x^2};
\addplot[draw=FigureGreen,line width=1.1pt,dashed,domain=-1:0,samples=60] {1-x-x^2};
\end{axis}
\begin{axis}[at={(A.east)},anchor=west,xshift=12mm,title={(b) 零延拓}]
\addplot[draw=FigureBlue,line width=1.1pt,domain=0:1,samples=60] {1+x-x^2};
\addplot[draw=FigureYellow,line width=1.1pt,dashed,domain=-1:0] {0};
\addplot[only marks,mark=o,mark size=1.4pt,draw=FigureYellow,fill=white] coordinates {(0,0)};
\addplot[only marks,mark=*,mark size=1.4pt,FigureBlue] coordinates {(0,1)};
\end{axis}
\end{tikzpicture}
