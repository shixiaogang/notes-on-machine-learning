% 第5章 5.2.3：核运算在截面上的效果，固定 x=0.4，t∈[-1,1]（解析式）。
% K1(t,x)=1+xt（仿射核），K2(t,x)=exp(-(t-x)^2/(2*0.3^2))（高斯核）。
% (a) 两个原始截面；(b) 和 K1+K2 与逐点乘积 K1*K2 的截面。
\begin{tikzpicture}[font=\figurefont\footnotesize]
\pgfplotsset{every axis/.style={width=52mm,height=38mm,axis lines=left,xmin=-1,xmax=1,
  xtick={-1,-0.5,0,0.4,1},xticklabels={$-1$,$-0.5$,$0$,$x$,$1$},ymin=0,ymax=2.5,ytick={0,1,2},
  x axis line style={draw=FigureStroke,line width=.6pt},y axis line style={draw=FigureStroke,line width=.6pt},
  tick style={draw=FigureStroke},tick label style={font=\figurefont\scriptsize,text=FigureInk},
  label style={font=\figurefont\scriptsize},xlabel={$t$},xlabel style={at={(axis description cs:1,0)},anchor=north west,yshift=2mm},
  ymajorgrids,grid style={draw=FigureGrid,line width=.4pt},
  title style={font=\figurefont\footnotesize},samples=161,domain=-1:1,
  legend style={draw=none,fill=none,font=\figurefont\scriptsize,at={(0.02,0.98)},anchor=north west,cells={anchor=west}}}}
\begin{axis}[name=A,title={(a) 原始核截面}]
\addplot[draw=FigureBlue,line width=1.1pt] {1+0.4*x};
\addplot[draw=FigureGreen,line width=1.1pt,dashed] {exp(-(x-0.4)^2/0.18)};
\legend{$K_1$：仿射,$K_2$：高斯}
\end{axis}
\begin{axis}[at={(A.east)},anchor=west,xshift=9mm,title={(b) 和与乘积}]
\addplot[draw=FigureYellow,line width=1.1pt] {1+0.4*x+exp(-(x-0.4)^2/0.18)};
\addplot[draw=FigurePurple,line width=1.1pt,dash dot] {(1+0.4*x)*exp(-(x-0.4)^2/0.18)};
\legend{$K_1+K_2$,$K_1K_2$}
\end{axis}
\end{tikzpicture}
