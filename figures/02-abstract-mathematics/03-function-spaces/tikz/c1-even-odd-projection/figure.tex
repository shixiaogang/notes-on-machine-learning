% 第1章 1.6.2：向偶函数子空间投影。[-1,1] 上 f(x)=e^x，Pf=cosh x（偶部），f-Pf=sinh x（奇部）。
% ||sinh||_2^2 = sinh(2)/2-1 ≈ 0.8134；奇部与每个偶函数正交。
\begin{tikzpicture}[font=\figurefont\footnotesize]
\begin{axis}[width=62mm,height=46mm,axis lines=middle,
  x axis line style={draw=FigureStroke,line width=.6pt,-{Stealth[length=2mm]}},
  y axis line style={draw=FigureStroke,line width=.6pt,-{Stealth[length=2mm]}},
  tick style={draw=FigureStroke},tick label style={font=\figurefont\scriptsize,text=FigureInk},
  xmin=-1.1,xmax=1.2,ymin=-1.4,ymax=3.0,xtick={-1,1},ytick={2},
  xlabel={$x$},xlabel style={anchor=north west,font=\figurefont\scriptsize},samples=100]
\addplot[draw=FigureBlue,line width=1.1pt,domain=-1:1] {exp(x)};
\addplot[draw=FigureGreen,line width=1.1pt,dashed,domain=-1:1] {cosh(x)};
\addplot[draw=FigureYellow,line width=1.1pt,domain=-1:1] {sinh(x)};
\node[font=\figurefont\scriptsize,text=FigureBlue,anchor=east] at (axis cs:0.6,2.3) {$f=e^x$};
\node[font=\figurefont\scriptsize,text=FigureGreen,anchor=south] at (axis cs:-0.55,1.5) {$Pf=\cosh x$};
\node[font=\figurefont\scriptsize,text=FigureYellow,anchor=north east] at (axis cs:1.0,-0.3) {$f-Pf=\sinh x$};
\end{axis}
\end{tikzpicture}
