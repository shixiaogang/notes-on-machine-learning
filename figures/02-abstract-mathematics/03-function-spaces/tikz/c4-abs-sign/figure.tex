% 第4章 4.2.2：f(x)=|x| 与其一阶弱导数 sgn x（解析式绘制）。原点处 sgn 的取值不影响弱导等价类，以空心点表示。
\begin{tikzpicture}[font=\figurefont\footnotesize]
\pgfplotsset{every axis/.style={width=50mm,height=34mm,axis lines=middle,
  x axis line style={draw=FigureStroke,line width=.6pt,-{Latex[length=1.6mm]}},
  y axis line style={draw=FigureStroke,line width=.6pt,-{Latex[length=1.6mm]}},
  tick style={draw=FigureStroke},tick label style={font=\figurefont\scriptsize,text=FigureInk},
  xlabel={$x$},xlabel style={anchor=west,font=\figurefont\scriptsize},
  title style={font=\figurefont\footnotesize,yshift=1mm},xmin=-1.3,xmax=1.4,xtick={-1,1}}}
\begin{axis}[name=A,ymin=-0.2,ymax=1.3,ytick={1},title={(a) $f(x)=|x|$}]
\addplot[draw=FigureBlue,line width=1.1pt,domain=-1.2:1.2,samples=3] {abs(x)};
\end{axis}
\begin{axis}[at={(A.east)},anchor=west,xshift=12mm,axis y line=left,xtick={-1,0,1},ymin=-1.4,ymax=1.4,ytick={-1,1},title={(b) 弱导数$\operatorname{sgn}x$}]
\addplot[draw=FigureBlue,line width=1.1pt,domain=-1.2:0] {-1};
\addplot[draw=FigureBlue,line width=1.1pt,domain=0:1.2] {1};
\addplot[only marks,mark=o,mark size=1.4pt,draw=FigureBlue,fill=white] coordinates {(0,-1) (0,1)};
\end{axis}
\end{tikzpicture}
