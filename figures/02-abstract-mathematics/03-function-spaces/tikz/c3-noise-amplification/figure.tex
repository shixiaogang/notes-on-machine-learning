% 第3章 3.4.4：第一类方程 Vf=g（(Vf)(x)=int_0^x f）中小奇异值放大扰动。
% 真解 f=1，数据 g=x；扰动 delta*u_k，u_k(x)=sqrt2 sin((k-1/2)pi x)，k=20，delta=0.01，
% ||delta u_k||=0.01。最小范数解的扰动为 (delta/sigma_k) v_k，sigma_k=1/((k-1/2)pi)=0.01632，
% v_k(t)=sqrt2 cos((k-1/2)pi t)，振幅 0.01*19.5*pi*sqrt2=0.8663，L^2 范数 0.6126。
\begin{tikzpicture}[font=\figurefont\footnotesize]
\pgfplotsset{every axis/.append style={width=54mm,height=42mm,xmin=0,xmax=1,domain=0:1,samples=500,
  axis x line=bottom,axis y line=left,xtick={0,0.5,1},
  tick label style={font=\figurefont\scriptsize,text=FigureInk},
  xlabel style={font=\figurefont\scriptsize},
  ymajorgrids,grid style={draw=FigureGrid,line width=.4pt},
  legend style={at={(0.03,0.97)},anchor=north west,font=\figurefont\scriptsize,cells={anchor=west}},
  title style={font=\figurefont\footnotesize}}}
\begin{axis}[name=A,ymin=-0.1,ymax=1.1,ytick={0,0.5,1},xlabel={$x$},
  title={(a) 数据：扰动范数$0.01$}]
\addplot[FigureBlue,line width=1.1pt] {x};
\addplot[FigureYellow,line width=.6pt] {x+0.01*sqrt(2)*sin(deg(19.5*pi*x))};
\legend{$g=Vf$,$g+\delta u_{20}$}
\end{axis}
\begin{axis}[at={(A.east)},anchor=west,xshift=14mm,ymin=-0.1,ymax=2.85,ytick={0,1,2},xlabel={$t$},legend columns=2,
  title={(b) 解：扰动范数约$0.61$}]
\addplot[FigureBlue,line width=1.1pt] {1};
\addplot[FigureYellow,line width=.6pt] {1+0.01*19.5*pi*sqrt(2)*cos(deg(19.5*pi*x))};
\legend{$f$,$f+(\delta/\sigma_{20})v_{20}$}
\end{axis}
\end{tikzpicture}
