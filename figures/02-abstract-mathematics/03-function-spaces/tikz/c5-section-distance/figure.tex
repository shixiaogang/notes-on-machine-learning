% 第5章 5.1.2：核截面距离 ||K_x-K_z||_H 随位置差 d=|x-z| 的变化（解析式）。
% 高斯核：sqrt(2-2exp(-d^2/(2 sigma^2)))，sigma=0.2 与 0.5；上界 d/sigma（sigma=0.5，点线）。
% 最小值核 min(x,z)（[0,1] 上）：sqrt(d)。离散核 1{x=z}：d>0 时恒为 sqrt2，d=0 时为 0。
\begin{tikzpicture}[font=\figurefont\footnotesize]
\begin{axis}[width=66mm,height=48mm,axis lines=left,xmin=0,xmax=1.02,ymin=0,ymax=1.65,
  xtick={0,0.25,0.5,0.75,1},ytick={0,0.5,1,1.414},yticklabels={$0$,$0.5$,$1$,$\sqrt2$},
  xlabel={$d=|x-z|$},ylabel={$\|K_x-K_z\|_H$},
  x axis line style={draw=FigureStroke,line width=.6pt},y axis line style={draw=FigureStroke,line width=.6pt},
  tick style={draw=FigureStroke},tick label style={font=\figurefont\scriptsize,text=FigureInk},
  label style={font=\figurefont\scriptsize},
  ymajorgrids,grid style={draw=FigureGrid,line width=.4pt},samples=161,domain=0:1,
  legend style={draw=none,fill=white,fill opacity=.9,text opacity=1,font=\figurefont\scriptsize,at={(1.03,0.5)},anchor=west,cells={anchor=west}}]
\addplot[draw=FigureBlue,line width=1.1pt] {sqrt(2-2*exp(-x^2/0.08))};
\addlegendentry{高斯核，$\sigma=0.2$}
\addplot[draw=FigureBlue,line width=1.1pt,dashed] {sqrt(2-2*exp(-2*x^2))};
\addlegendentry{高斯核，$\sigma=0.5$}
\addplot[draw=FigureMuted,line width=.7pt,densely dotted,domain=0:0.8] {2*x};
\addlegendentry{上界$d/\sigma$，$\sigma=0.5$}
\addplot[draw=FigureGreen,line width=1.1pt,dash dot] {sqrt(x)};
\addlegendentry{最小值核}
\addplot[draw=FigureRed,line width=1.1pt,domain=0.012:1] {1.41421};
\addlegendentry{离散核}
\addplot[only marks,mark=*,mark size=1.4pt,FigureRed,forget plot] coordinates {(0,0)};
\addplot[only marks,mark=o,mark size=1.4pt,FigureRed,fill=white,forget plot] coordinates {(0,1.41421)};
\end{axis}
\end{tikzpicture}
