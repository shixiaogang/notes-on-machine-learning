% 第4章 4.5.4：Poincaré 不等式中去除常数方向的对照（解析式绘制）。
% (a) 常数 u=1：u'=0 而 ||u||_2=1，无约束时不等式失败；(b) 零边界 sin(pi x)：||u||_2/||u'||_2=1/pi；
% (c) 零均值 cos(pi x)：同样比值 1/pi；(d) 两个分支 (0,1)∪(2,3) 上取 -1 与 1：零均值、导数为零，连通性不可少。
\begin{tikzpicture}[font=\figurefont\footnotesize]
\pgfplotsset{every axis/.style={width=38mm,height=30mm,axis lines=middle,
  x axis line style={draw=FigureStroke,line width=.6pt},y axis line style={draw=FigureStroke,line width=.6pt},
  tick style={draw=FigureStroke},tick label style={font=\figurefont\scriptsize,text=FigureInk},
  title style={font=\figurefont\footnotesize,yshift=1mm},ymin=-1.3,ymax=1.3,ytick={-1,1},samples=100,axis on top}}
\begin{axis}[name=A,xmin=0,xmax=1.1,xtick={1},title={(a) 常数}]
\addplot[draw=FigureYellow,line width=1.1pt,domain=0:1] {1};
\end{axis}
\begin{axis}[name=B,at={(A.east)},anchor=west,xshift=6mm,xmin=0,xmax=1.1,xtick={1},title={(b) 零边界}]
\addplot[draw=FigureBlue,line width=1.1pt,domain=0:1] {sin(deg(pi*x))};
\addplot[only marks,mark=*,mark size=1.3pt,FigureBlue] coordinates {(0,0) (1,0)};
\end{axis}
\begin{axis}[name=C,at={(B.east)},anchor=west,xshift=6mm,xmin=0,xmax=1.1,xtick={1},title={(c) 零均值}]
\addplot[draw=none,fill=FigureBlueFill,domain=0:1] {cos(deg(pi*x))} \closedcycle;
\addplot[draw=FigureBlue,line width=1.1pt,domain=0:1] {cos(deg(pi*x))};
\end{axis}
\begin{axis}[at={(C.east)},anchor=west,xshift=6mm,xmin=0,xmax=3.2,xtick={1,2,3},title={(d) 两个分支}]
\addplot[draw=FigureYellow,line width=1.1pt,domain=0:1] {-1};
\addplot[draw=FigureYellow,line width=1.1pt,domain=2:3] {1};
\end{axis}
\end{tikzpicture}
