% 第1章 1.5.2：换测度后的平方积分。(0,1] 上 f(x)=x^{-1/2}，密度 w(x)=2x。
% |f|^2=1/x 在 Lebesgue 测度下积分发散；|f|^2 w = 2，在 nu=w dx 下积分为 2。
\begin{tikzpicture}[font=\figurefont\footnotesize]
\begin{axis}[width=64mm,height=46mm,axis lines=left,
  x axis line style={draw=FigureStroke,line width=.6pt,-{Stealth[length=2mm]}},
  y axis line style={draw=FigureStroke,line width=.6pt,-{Stealth[length=2mm]}},
  tick style={draw=FigureStroke},tick label style={font=\figurefont\scriptsize,text=FigureInk},
  xmin=0,xmax=1.12,ymin=0,ymax=6.3,xtick={0,0.5,1},ytick={0,2,4,6},
  xlabel={$x$},xlabel style={at={(axis description cs:1,0)},anchor=north west,yshift=1mm,font=\figurefont\scriptsize},
  ymajorgrids,grid style={draw=FigureGrid,line width=.4pt},samples=200]
\addplot[fill=FigureBlueFill,draw=none,domain=0.16:1] {1/x} \closedcycle;
\addplot[draw=FigureBlue,line width=1.1pt,domain=0.16:1] {1/x};
\addplot[draw=FigureYellow,line width=1.1pt,domain=0:1] {2*x};
\addplot[draw=FigureGreen,line width=1.1pt,domain=0:1] {2};
\node[font=\figurefont\scriptsize,text=FigureBlue,anchor=west] at (axis cs:0.20,5.4) {$|f|^2=1/x$};
\node[font=\figurefont\scriptsize,text=FigureYellow,anchor=north west] at (axis cs:0.60,0.95) {$w=2x$};
\node[font=\figurefont\scriptsize,text=FigureGreen,anchor=west] at (axis cs:0.55,2.45) {$|f|^2w=2$};
\end{axis}
\end{tikzpicture}
