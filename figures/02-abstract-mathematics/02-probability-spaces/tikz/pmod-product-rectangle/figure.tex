% X,Y独立且都服从Unif[0,1)，联合分布为单位正方形上的面积（乘积测度）。
% (a) 矩形A x B，A=[0.2,0.6)，B=[0.5,0.9)：P(X in A, Y in B)=0.4*0.4=0.16，等于两条边缘概率之积。
% (b) 非矩形事件{X<Y}：按x切片，切片长度1-x，Tonelli给出概率int_0^1 (1-x)dx=1/2。
\begin{tikzpicture}[line width=.6pt,font=\fontsize{8}{10}\selectfont,>=Stealth]
\def\s{32mm}
% ---------- (a) ----------
\begin{scope}
 \draw[FigureStroke] (0,0) rectangle (\s,\s);
 \fill[FigureBlueFill,draw=FigureBlue] (.2*\s,.5*\s) rectangle (.6*\s,.9*\s);
 \node[FigureInk] at (.4*\s,.7*\s) {$A\times B$};
 \draw[FigureYellow,line width=2pt] (.2*\s,-1.6mm) -- (.6*\s,-1.6mm);
 \draw[FigureGreen,line width=2pt] (-1.6mm,.5*\s) -- (-1.6mm,.9*\s);
 \node[FigureInk] at (.4*\s,-4.6mm) {$A$};
 \node[FigureInk] at (-4.6mm,.7*\s) {$B$};
 \draw[FigureGrid,dashed,line width=.4pt] (.2*\s,0) -- (.2*\s,.5*\s) (.6*\s,0) -- (.6*\s,.5*\s) (0,.5*\s) -- (.2*\s,.5*\s) (0,.9*\s) -- (.2*\s,.9*\s);
 \foreach \t in {0,1}{\node[FigureMuted,font=\fontsize{7.5}{9}\selectfont,anchor=north] at (\t*\s,-.5mm) {$\t$};}
 \node[FigureMuted,font=\fontsize{7.5}{9}\selectfont,anchor=east] at (-.5mm,\s) {$1$};
 \node[FigureInk,anchor=west] at (\s+1mm,0) {$x$};
 \node[FigureInk,anchor=south] at (0,\s+.5mm) {$y$};
 \node[FigureInk] at (.5*\s,-10mm) {(a) $0.4\times0.4=0.16$};
\end{scope}
% ---------- (b) ----------
\begin{scope}[xshift=56mm]
 \draw[FigureStroke] (0,0) rectangle (\s,\s);
 \fill[FigureGreenFill,draw=FigureGreen] (0,0) -- (\s,\s) -- (0,\s) -- cycle;
 \node[FigureInk] at (.22*\s,.62*\s) {$\{X<Y\}$};
 \draw[FigureRed,line width=1.1pt] (.62*\s,.62*\s) -- (.62*\s,\s);
 \draw[FigureRed,dashed,line width=.4pt] (.62*\s,0) -- (.62*\s,.62*\s);
 \node[FigureRed,font=\fontsize{7.5}{9}\selectfont,anchor=east] at (.60*\s,.86*\s) {$1-x$};
 \node[FigureMuted,font=\fontsize{7.5}{9}\selectfont,anchor=north] at (.62*\s,-.5mm) {$x$};
 \foreach \t in {0,1}{\node[FigureMuted,font=\fontsize{7.5}{9}\selectfont,anchor=north] at (\t*\s,-.5mm) {$\t$};}
 \node[FigureMuted,font=\fontsize{7.5}{9}\selectfont,anchor=east] at (-.5mm,\s) {$1$};
 \node[FigureInk,anchor=west] at (\s+1mm,0) {$x$};
 \node[FigureInk,anchor=south] at (0,\s+.5mm) {$y$};
 \node[FigureInk] at (.5*\s,-10mm) {(b) $\int_0^1(1-x)\,\mathrm dx=1/2$};
\end{scope}
\end{tikzpicture}
