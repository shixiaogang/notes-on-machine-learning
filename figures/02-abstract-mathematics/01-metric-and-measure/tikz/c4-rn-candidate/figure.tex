% Radon–Nikodym证明的示意：[0,1]上目标测度的（事后可知的）密度f(x)=1.2+0.6sin(2πx)，候选h=min(f,1)，剩余质量为f-h；eps=0.25，P={f-h>=eps}即sin(2πx)>=1/12，约为x∈[0.0133,0.4867]；h+eps·1_P仍在f之下。横向1单位=8cm，纵向1单位=1.6cm。
\begin{tikzpicture}[
  lab/.style={font=\figurefont\fontsize{8}{10}\selectfont},
  note/.style={font=\FigureNoteFont\fontsize{7.5}{9}\selectfont,text=FigureMuted}]
\def\sx{8}\def\sy{1.6}
\def\pa{0.0133}\def\pb{0.4867}
% residual region between h and f
\fill[FigureYellowFill,domain=0:1,samples=120,variable=\t]
  plot ({\t*\sx},{max(1.2+0.6*sin(360*\t),1)*\sy}) -- (\sx,\sy) -- (0,\sy) -- cycle;
% candidate area
\fill[FigureBlueFill,domain=0:1,samples=120,variable=\t] (0,0) -- plot ({\t*\sx},{min(1.2+0.6*sin(360*\t),1)*\sy}) -- (\sx,0) -- cycle;
% epsilon strip on P
\fill[FigureGreenFill] (\pa*\sx,\sy) rectangle (\pb*\sx,1.25*\sy);
\draw[FigureGreen,line width=1.1pt] (\pa*\sx,1.25*\sy)--(\pb*\sx,1.25*\sy);
\draw[FigureGreen,line width=.6pt] (\pa*\sx,\sy)--(\pa*\sx,1.25*\sy) (\pb*\sx,\sy)--(\pb*\sx,1.25*\sy);
\draw[->,FigureStroke,line width=.7pt] (-0.1,0)--(1.06*\sx,0) node[lab,right] {$x$};
\draw[->,FigureStroke,line width=.7pt] (0,-0.1)--(0,2.15*\sy);
\draw[FigureRed,line width=1.1pt,dashed,domain=0:1,samples=120,variable=\t] plot ({\t*\sx},{(1.2+0.6*sin(360*\t))*\sy});
\draw[FigureBlue,line width=1.1pt,domain=0:1,samples=120,variable=\t] plot ({\t*\sx},{min(1.2+0.6*sin(360*\t),1)*\sy});
\foreach \x/\t in {0/0,0.5/\frac12,1/1}{\draw[FigureStroke] (\x*\sx,0)--(\x*\sx,-0.06); \node[lab,below] at (\x*\sx,-0.04) {$\t$};}
\foreach \y/\t in {1/1,1.25/{1{+}\varepsilon}}{\draw[FigureStroke] (0,\y*\sy)--(-0.06,\y*\sy);}
\node[lab,left] at (-0.04,1*\sy) {$1$};
\node[lab,left] at (-0.04,1.27*\sy) {$1+\varepsilon$};
\draw[FigureGreen,line width=1.4pt] (\pa*\sx,-0.62)--(\pb*\sx,-0.62);
\node[lab,text=FigureGreen] at (0.25*\sx,-0.88) {$P$};
\node[lab,text=FigureRed] at (0.25*\sx,1.95*\sy) {$f$};
\node[lab,text=FigureBlue] at (0.75*\sx,0.82*\sy) {$h$};
\node[note] at (0.5*\sx,0.45*\sy) {已被候选$h$表示的质量};
\node[note] at (0.25*\sx,1.52*\sy) {剩余质量$r$};
\node[note,anchor=west] at (0.52*\sx,1.25*\sy) {$\leftarrow$ 在$P$上加高$\varepsilon$};
\end{tikzpicture}
