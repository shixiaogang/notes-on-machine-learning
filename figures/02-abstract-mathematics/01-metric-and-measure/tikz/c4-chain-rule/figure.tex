% 密度的链式规则：nu<<mu<<rho，dnu/drho=(dnu/dmu)(dmu/drho)。数值示例：[0,1]上rho=lambda，dmu/dlambda=2x，dnu/dmu=3x/2，于是dnu/dlambda=3x^2；三者总质量都为1。
\begin{tikzpicture}[>=Stealth,
  m/.style={draw=FigureBlue,fill=FigureBlueFill,rounded corners=2pt,line width=.7pt,minimum width=12mm,minimum height=7mm,font=\figurefont\fontsize{8}{10}\selectfont},
  a/.style={draw=FigureStroke,line width=.7pt,->,shorten >=2pt,shorten <=2pt},
  el/.style={font=\FigureNoteFont\fontsize{7.5}{9}\selectfont,text=FigureInk},
  note/.style={font=\FigureNoteFont\fontsize{7.5}{9}\selectfont,text=FigureMuted}]
\node[m] (r) at (0,0) {$\rho$};
\node[m] (mu) at (3.6,0) {$\mu$};
\node[m] (nu) at (7.2,0) {$\nu$};
\draw[a] (r)--(mu) node[el,midway,above] {$\dfrac{\mathrm d\mu}{\mathrm d\rho}$};
\draw[a] (mu)--(nu) node[el,midway,above] {$\dfrac{\mathrm d\nu}{\mathrm d\mu}$};
\draw[a,draw=FigureGreen,line width=.9pt] (r.south) to[bend right=22] node[el,midway,below] {$\dfrac{\mathrm d\nu}{\mathrm d\rho}=\dfrac{\mathrm d\nu}{\mathrm d\mu}\cdot\dfrac{\mathrm d\mu}{\mathrm d\rho}$\quad $\rho$-a.e.} (nu.south);
\node[note] at (1.8,-0.55) {$2x$};
\node[note] at (5.4,-0.55) {$\frac32x$};
\node[note,text=FigureGreen] at (3.6,-2.25) {例：$\rho=\lambda$时，$\frac32x\cdot2x=3x^2$};
\end{tikzpicture}
