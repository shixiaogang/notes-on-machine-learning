% 二进分层逼近 f(x)=x^2（[0,2]上）：s_n=2^{-n}floor(2^n min(f,n))。
% n=1：断点 sqrt(1/2)=0.7071、1，取值 0,0.5,1（截断高度1）；n=2：断点 sqrt(k/4)，k=1..8，取值 0,0.25,...,2（截断高度2）。
\begin{tikzpicture}[font=\fontsize{8}{10}\selectfont]
\pgfplotsset{every axis/.style={width=58mm,height=48mm,xmin=0,xmax=2.08,ymin=0,ymax=4.3,
  axis lines=left,axis line style={FigureStroke,line width=.75pt,-{Stealth[length=4pt]}},
  tick style={FigureStroke,line width=.6pt},tick label style={text=FigureInk},
  xtick={0,0.5,1,1.5,2},ytick={0,1,2,3,4},xlabel={$x$},
  xlabel style={at={(ticklabel* cs:1)},anchor=north west,text=FigureInk},
  ylabel style={rotate=-90,at={(ticklabel* cs:1)},anchor=south,text=FigureInk},
  ymajorgrids,grid style={FigureGrid,line width=.4pt}}}
\begin{axis}[name=a,title={(a) $n=1$：步长$1/2$，截断于$1$}]
  \addplot[draw=FigureBlue,fill=FigureBlueFill,line width=.9pt,const plot] coordinates
    {(0,0) (0.7071,0.5) (1,1) (2,1)} \closedcycle;
  \addplot[FigureStroke,dashed,line width=.6pt,domain=0:2.08] {1};
  \addplot[FigureRed,line width=1.1pt,domain=0:2,samples=80] {x^2};
  \node[text=FigureRed,anchor=west] at (axis cs:1.42,3.5) {$f=x^2$};
  \node[text=FigureBlue] at (axis cs:1.55,0.55) {$s_1$};
\end{axis}
\begin{axis}[at={(a.east)},anchor=west,xshift=8mm,title={(b) $n=2$：步长$1/4$，截断于$2$}]
  \addplot[draw=FigureBlue,fill=FigureBlueFill,line width=.9pt,const plot] coordinates
    {(0,0) (0.5,0.25) (0.7071,0.5) (0.8660,0.75) (1,1) (1.1180,1.25) (1.2247,1.5) (1.3229,1.75) (1.4142,2) (2,2)} \closedcycle;
  \addplot[FigureStroke,dashed,line width=.6pt,domain=0:2.08] {2};
  \addplot[FigureRed,line width=1.1pt,domain=0:2,samples=80] {x^2};
  \node[text=FigureRed,anchor=west] at (axis cs:1.42,3.5) {$f=x^2$};
  \node[text=FigureBlue] at (axis cs:1.72,1.2) {$s_2$};
\end{axis}
\end{tikzpicture}
