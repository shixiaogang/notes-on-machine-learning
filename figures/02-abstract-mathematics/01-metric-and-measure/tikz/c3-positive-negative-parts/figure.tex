% 正负部分解：f(x)=sin x，x∈[0,2π]；f^+=max(f,0) 在 [0,π] 上等于 sin x，f^-=max(-f,0) 在 [π,2π] 上等于 -sin x；|f|=f^++f^-。
\begin{tikzpicture}[font=\fontsize{8}{10}\selectfont]
\pgfplotsset{every axis/.style={width=56mm,height=40mm,xmin=0,xmax=6.9,ymin=-1.25,ymax=1.35,
  axis lines=middle,axis line style={FigureStroke,line width=.75pt,-{Stealth[length=4pt]}},
  tick style={FigureStroke,line width=.6pt},tick label style={text=FigureInk},
  xtick={3.14159,6.28319},xticklabels={$\pi$,$2\pi$},ytick={-1,1},
  xlabel={$x$},xlabel style={anchor=west,text=FigureInk},
  title style={yshift=1mm}}}
\begin{axis}[name=a,title={(a) $f=\sin x$}]
  \addplot[draw=none,fill=FigureBlueFill,domain=0:3.14159,samples=60] {sin(deg(x))} \closedcycle;
  \addplot[draw=none,fill=FigureRedFill,domain=3.14159:6.28319,samples=60] {sin(deg(x))} \closedcycle;
  \addplot[FigureStroke,line width=1.1pt,domain=0:6.28319,samples=120] {sin(deg(x))};
  \node[text=FigureBlue] at (axis cs:1.57,0.45) {$f>0$};
  \node[text=FigureRed] at (axis cs:4.71,-0.45) {$f<0$};
\end{axis}
\begin{axis}[at={(a.east)},anchor=west,xshift=8mm,ymin=-0.25,ymax=1.35,ytick={1},title={(b) 正部$f^+$与负部$f^-$}]
  \addplot[draw=FigureBlue,fill=FigureBlueFill,line width=1.1pt,domain=0:3.14159,samples=60] {sin(deg(x))} \closedcycle;
  \addplot[draw=FigureRed,fill=FigureRedFill,line width=1.1pt,domain=3.14159:6.28319,samples=60] {-sin(deg(x))} \closedcycle;
  \node[text=FigureBlue] at (axis cs:1.57,1.17) {$f^+$};
  \node[text=FigureRed] at (axis cs:4.71,1.17) {$f^-$};
\end{axis}
\end{tikzpicture}
