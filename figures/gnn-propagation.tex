% 原创教学示意；每一轮同步读取前一轮状态。
\begin{tikzpicture}[x={\dimexpr\linewidth/112\relax},y=1mm,
  font=\figurefont\footnotesize,text=NNInk]
  \foreach \l/\xx in {0/12,1/55,2/98}{
    \node[nn heading] at (\xx,31) {第$\l$轮};
    \foreach \i/\yy in {1/18,2/0,3/-18}{
      \ifnum\l=0
        \node[nn input,minimum size=10mm] (h\l\i) at (\xx,\yy) {$h_\i^{(\l)}$};
      \else
        \node[nn hidden,minimum size=10mm] (h\l\i) at (\xx,\yy) {$h_\i^{(\l)}$};
      \fi
    }
  }
  \foreach \a/\b in {0/1,1/2}{
    \foreach \i in {1,2,3}{\draw[nn operator path] (h\a\i)--(h\b\i);}
    \draw[nn flow,draw=NNLine] (h\a1)--(h\b2);
    \draw[nn flow,draw=NNLine] (h\a2)--(h\b1);
    \draw[nn flow,draw=NNLine] (h\a2)--(h\b3);
    \draw[nn flow,draw=NNLine] (h\a3)--(h\b2);
  }
  \draw[nn flow,draw=NNOutput,line width=1.4pt] (h03)--(h12);
  \draw[nn flow,draw=NNOutput,line width=1.4pt] (h12)--(h21);
\end{tikzpicture}
