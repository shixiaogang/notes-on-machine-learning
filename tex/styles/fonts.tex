% build.sh 将必需的中文字体解析到 build/fonts；缺失时报错，不静默替换字形。
\setmainfont{STIXTwoText}[
  Extension=.otf,
  UprightFont=*-Regular, BoldFont=*-Bold,
  ItalicFont=*-Italic, BoldItalicFont=*-BoldItalic]
\setsansfont{SourceSans3}[
  Extension=.otf,
  UprightFont=*-Regular, BoldFont=*-Semibold,
  ItalicFont=*-RegularIt, BoldItalicFont=*-SemiboldIt,
  Scale=MatchLowercase]
\setmonofont{SourceCodePro}[
  Extension=.otf,
  UprightFont=*-Regular, BoldFont=*-Semibold,
  ItalicFont=*-RegularIt, BoldItalicFont=*-SemiboldIt,
  Scale=0.88]
\setCJKmainfont{SourceHanSerifCN-Regular.otf}[
  Path=build/fonts/, BoldFont={SourceHanSerifCN-Bold.otf},
  ItalicFont={LXGWWenKai-Regular.ttf}, BoldItalicFont={LXGWWenKai-Medium.ttf}]
\setCJKsansfont{SourceHanSansSC-Regular.otf}[
  Path=build/fonts/, BoldFont={SourceHanSansSC-Bold.otf}]
\setCJKmonofont{LXGWWenKaiMono-Regular.ttf}[Path=build/fonts/]
\setCJKfamilyfont{bookkai}{LXGWWenKai-Regular.ttf}[
  Path=build/fonts/, BoldFont={LXGWWenKai-Medium.ttf},
  ItalicFont={LXGWWenKai-Regular.ttf}, BoldItalicFont={LXGWWenKai-Medium.ttf}]
